// Gallery automatisch genereren met exacte bestandsnamen const imageFiles = ["Foto1.JPG", "Foto2.JPG", "Foto3.JPG"]; const gallery = document.getElementById('gallery'); imageFiles.forEach(file => { const src = `assets/img/evenementen/${file}`; const a = document.createElement('a'); a.href = src; a.setAttribute('data-fancybox','evenementen'); a.setAttribute('data-caption', file); const img = document.createElement('img'); img.src = src; img.alt = file; a.appendChild(img); gallery.appendChild(a); });
